SELECT SUM(AMOUNT) AS TOTAL_DEPOSIT
FROM DEPOSIT
WHERE ADATE > TO_DATE('01-JAN-1996','DD-MON-YYYY');

SELECT SUM(D.AMOUNT) AS TOTAL_DEPOSIT
FROM DEPOSIT D

SELECT MAX(D.AMOUNT) AS MAX_DEPOSIT
FROM DEPOSIT D

SELECT ROUND(MAX(EMP_SAL)) AS MAXIMUM,
       ROUND(MIN(EMP_SAL)) AS MINIMUM,
       ROUND(SUM(EMP_SAL)) AS SUM,
       ROUND(AVG(EMP_SAL)) AS AVERAGE
FROM EMPLOYEE;

SELECT MAX(EMP_SAL) - MIN(EMP_SAL) AS DIFFERENCE
FROM EMPLOYEE;

SELECT COUNT(*) AS TOTAL_EMPLOYEES,
       SUM(CASE WHEN EXTRACT(YEAR FROM HIRE_DATE) = 1995 THEN 1 ELSE 0 END) AS HIRED_1995,
       SUM(CASE WHEN EXTRACT(YEAR FROM HIRE_DATE) = 1996 THEN 1 ELSE 0 END) AS HIRED_1996,
       SUM(CASE WHEN EXTRACT(YEAR FROM HIRE_DATE) = 1997 THEN 1 ELSE 0 END) AS HIRED_1997,
       SUM(CASE WHEN EXTRACT(YEAR FROM HIRE_DATE) = 1998 THEN 1 ELSE 0 END) AS HIRED_1998
FROM EMPLOYEE;

SELECT ROUND(AVG(EMP_SAL),2) AS AVERAGE_SALARY
FROM EMPLOYEE
GROUP BY DEPT_NO;

SELECT D.DEPT_NO,
       J.JOB_TITLE,
       SUM(E.EMP_SAL) AS TOTAL_SALARY
FROM EMPLOYEE E
JOIN EMP_JOB EJ ON E.EMP_NO = EJ.EMP_NO
JOIN JOB J ON EJ.JOB_ID = J.JOB_ID
JOIN DEPARTMENT D ON E.DEPT_NO = D.DEPT_NO
GROUP BY D.DEPT_NO, J.JOB_TITLE;

SELECT DEPT_NO,
       ROUND(AVG(EMP_SAL),2) AS AVG_SALARY
FROM EMPLOYEE
GROUP BY DEPT_NO
HAVING AVG(EMP_SAL) > 2000;

SELECT J.JOB_TITLE,
       SUM(E.EMP_SAL) AS TOTAL_SALARY
FROM EMPLOYEE E
JOIN EMP_JOB EJ ON E.EMP_NO = EJ.EMP_NO
JOIN JOB J ON EJ.JOB_ID = J.JOB_ID
WHERE UPPER(J.JOB_TITLE) <> 'PRESIDENT'
GROUP BY J.JOB_TITLE
HAVING SUM(E.EMP_SAL) > 3000
ORDER BY TOTAL_SALARY;

SELECT B.BNAME,

FROM BRANCH B
JOIN DEPOSIT D ON B.BNAME = D.BNAME
WHERE B.CITY = 'BOMBAY'
GROUP BY B.BNAME
HAVING SUM(D.AMOUNT) > 5000;

-- 1. Add 2 Numbers
SET SERVEROUTPUT ON;

DECLARE
    A NUMBER := 10;
    B NUMBER := 20;
    C NUMBER;
BEGIN
    C := A + B;
    DBMS_OUTPUT.PUT_LINE('SUM = ' || C);
END;
/

-- 2(a). Area of Rectangle
SET SERVEROUTPUT ON;

DECLARE
    L NUMBER := 10;
    B NUMBER := 5;
    AREA NUMBER;
BEGIN
    AREA := L * B;
    DBMS_OUTPUT.PUT_LINE('AREA OF RECTANGLE = ' || AREA);
END;
/

-- 2(b). Area of Triangle
SET SERVEROUTPUT ON;

DECLARE
    BASE NUMBER := 10;
    HEIGHT NUMBER := 5;
    AREA NUMBER;
BEGIN
    AREA := 0.5 * BASE * HEIGHT;
    DBMS_OUTPUT.PUT_LINE('AREA OF TRIANGLE = ' || AREA);
END;
/

-- 2(c). Area of Square
SET SERVEROUTPUT ON;

DECLARE
    SIDE NUMBER := 8;
    AREA NUMBER;
BEGIN
    AREA := SIDE * SIDE;
    DBMS_OUTPUT.PUT_LINE('AREA OF SQUARE = ' || AREA);
END;
/

-- 3. Maximum of 3 Numbers
SET SERVEROUTPUT ON;

DECLARE
    A NUMBER := 10;
    B NUMBER := 20;
    C NUMBER := 15;
BEGIN
    IF A >= B AND A >= C THEN
        DBMS_OUTPUT.PUT_LINE('MAXIMUM = ' || A);
    ELSIF B >= A AND B >= C THEN
        DBMS_OUTPUT.PUT_LINE('MAXIMUM = ' || B);
    ELSE
        DBMS_OUTPUT.PUT_LINE('MAXIMUM = ' || C);
    END IF;
END;
/

-- 4. Sum of N Numbers using FOR Loop
SET SERVEROUTPUT ON;

DECLARE
    N NUMBER := 10;
    SUM1 NUMBER := 0;
BEGIN
    FOR I IN 1..N LOOP
        SUM1 := SUM1 + I;
    END LOOP;
    DBMS_OUTPUT.PUT_LINE('SUM = ' || SUM1);
END;
/

-- 5. Fibonacci Series of N Numbers
SET SERVEROUTPUT ON;

DECLARE
    N NUMBER := 10;
    A NUMBER := 0;
    B NUMBER := 1;
    C NUMBER;
BEGIN
    DBMS_OUTPUT.PUT_LINE(A);
    DBMS_OUTPUT.PUT_LINE(B);

    FOR I IN 3..N LOOP
        C := A + B;
        DBMS_OUTPUT.PUT_LINE(C);
        A := B;
        B := C;
    END LOOP;
END;
/
